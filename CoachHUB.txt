local PoseState = {
    isPoseActive = false,
    poseConnection = nil
}

local BAT_POSE = {
    Torso = {rotation = CFrame.Angles(math.rad(75), math.rad(180), math.rad(0))},
    Neck = {rotation = CFrame.Angles(math.rad(45), math.rad(180), math.rad(0))},
    RightShoulder = {rotation = CFrame.Angles(math.rad(-90), math.rad(0), math.rad(0))},
    LeftShoulder = {rotation = CFrame.Angles(math.rad(90), math.rad(0), math.rad(0))},
    RightHip = {rotation = CFrame.Angles(math.rad(-60), math.rad(0), math.rad(0))},
    LeftHip = {rotation = CFrame.Angles(math.rad(60), math.rad(0), math.rad(0))}
}

local function applyBatPose(char)
    if not char or not char.Parent then return end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hrp then return end
    
    PoseState.isPoseActive = true
    
    for partName, poseData in pairs(BAT_POSE) do
        local part = char:FindFirstChild(partName)
        if part then
            TweenService:Create(part, TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
                {CFrame = hrp.CFrame * poseData.rotation}):Play()
        end
    end
    
    if PoseState.poseConnection then PoseState.poseConnection:Disconnect() end
    PoseState.poseConnection = RunService.RenderStepped:Connect(function()
        if not PoseState.isPoseActive or not char or not char.Parent then
            if PoseState.poseConnection then PoseState.poseConnection:Disconnect() end
            return
        end
        pcall(function()
            if hrp then
                for partName, poseData in pairs(BAT_POSE) do
                    local part = char:FindFirstChild(partName)
                    if part then
                        part.CFrame = hrp.CFrame * poseData.rotation
                    end
                end
            end
        end)
    end)
end

local function resetBatPose(char)
    if not char then return end
    PoseState.isPoseActive = false
    if PoseState.poseConnection then PoseState.poseConnection:Disconnect() end
    local h = char:FindFirstChildOfClass("Humanoid")
    if h then h:MoveTo(h:FindFirstChild("HumanoidRootPart") and h:FindFirstChild("HumanoidRootPart").Position) end
end

local function coachySaveOriginalLighting()
    if coachyOriginalLighting then return end
    coachyOriginalLighting = {
        ClockTime = Lighting.ClockTime,
        OutdoorAmbient = Lighting.OutdoorAmbient,
        Ambient = Lighting.Ambient,
        Brightness = Lighting.Brightness,
        FogStart = Lighting.FogStart,
        FogEnd = Lighting.FogEnd,
        FogColor = Lighting.FogColor,
        ColorShift_Top = Lighting.ColorShift_Top,
        ColorShift_Bottom = Lighting.ColorShift_Bottom,
        GeographicLatitude = Lighting.GeographicLatitude,
        GlobalShadows = Lighting.GlobalShadows,
        LightingChildren = {},
        TerrainChildren = {}
    }
    for _,child in ipairs(Lighting:GetChildren()) do
        if child:IsA("Sky") or child:IsA("Atmosphere") then 
            table.insert(coachyOriginalLighting.LightingChildren, child:Clone()) 
        end
    end
    local terrain = workspace:FindFirstChildOfClass("Terrain")
    if terrain then
        for _,child in ipairs(terrain:GetChildren()) do
            if child:IsA("Clouds") then 
                table.insert(coachyOriginalLighting.TerrainChildren, child:Clone()) 
            end
        end
    end
end

local function coachyClearSky()
    for _,child in ipairs(Lighting:GetChildren()) do
        if child:GetAttribute(CANDY_SKY_TAG) or child:IsA("Sky") or child:IsA("Atmosphere") then 
            pcall(function() child:Destroy() end) 
        end
    end
    local terrain = workspace:FindFirstChildOfClass("Terrain")
    if terrain then
        for _,child in ipairs(terrain:GetChildren()) do
            if child:GetAttribute(CANDY_SKY_TAG) or child:IsA("Clouds") then 
                pcall(function() child:Destroy() end) 
            end
        end
    end
end

local function coachyInstance(className,parent,props)
    local inst = Instance.new(className)
    inst:SetAttribute(CANDY_SKY_TAG, true)
    for k,v in pairs(props or {}) do 
        pcall(function() inst[k] = v end) 
    end
    inst.Parent = parent
    return inst
end

local function coachyColor(rgb)
    return Color3.fromRGB(rgb[1], rgb[2], rgb[3])
end

local function ApplyCoachySky()
    coachySaveOriginalLighting()
    coachyClearSky()
    local terrain = workspace:FindFirstChildOfClass("Terrain")
    local preset = COACHY_SKY_PRESET
    
    -- Semi-transparent white sky
    Lighting.FogStart = 200
    Lighting.FogEnd = 8000
    Lighting.FogColor = coachyColor({255,255,255})
    Lighting.ColorShift_Top = coachyColor({200,200,220})
    Lighting.ColorShift_Bottom = coachyColor({220,220,235})
    Lighting.GlobalShadows = false
    Lighting.ClockTime = preset.clock or 12
    Lighting.Brightness = preset.brightness or 4.5
    Lighting.OutdoorAmbient = coachyColor(preset.outAmb)
    Lighting.Ambient = coachyColor(preset.ambient)
    Lighting.ExposureCompensation = 0
    Lighting.EnvironmentDiffuseScale = 0
    Lighting.EnvironmentSpecularScale = 0
    
    -- Sky with no sun/moon
    local skyProps = {
        StarCount = 0,
        MoonAngularSize = 0,
        SunAngularSize = 0,
        SkyboxBk = "rbxassetid://0",
        SkyboxDn = "rbxassetid://0",
        SkyboxFt = "rbxassetid://0",
        SkyboxLf = "rbxassetid://0",
        SkyboxRt = "rbxassetid://0",
        SkyboxUp = "rbxassetid://0"
    }
    coachyInstance("Sky", Lighting, skyProps)
    
    -- Semi-transparent atmosphere
    coachyInstance("Atmosphere", Lighting, {
        Density = preset.atm.dens or 0.55,
        Color = coachyColor(preset.atm.color),
        Decay = coachyColor(preset.atm.decay),
        Glare = preset.atm.glare or 1.5,
        Haze = preset.atm.haze or 1.5,
        Offset = 0,
        Scale = 0.7
    })
    
    -- Light clouds
    if terrain then
        coachyInstance("Clouds", terrain, {
            Cover = preset.clouds.cover or 0.6,
            Density = preset.clouds.dens or 0.5,
            Color = coachyColor(preset.clouds.color)
        })
    end
end

local function RemoveCoachySky()
    if coachyOriginalLighting then
        for k,v in pairs(coachyOriginalLighting) do
            if k ~= "LightingChildren" and k ~= "TerrainChildren" then 
                pcall(function() Lighting[k] = v end) 
            end
        end
        for _,child in ipairs(coachyOriginalLighting.LightingChildren or {}) do 
            child:Clone().Parent = Lighting 
        end
        local offTerrain = workspace:FindFirstChildOfClass("Terrain")
        if offTerrain then
            for _,child in ipairs(coachyOriginalLighting.TerrainChildren or {}) do 
                child:Clone().Parent = offTerrain 
            end
        end
    end
    coachyClearSky()
end

local function makeDraggable(frame,handle,onEnd)
    local src=handle or frame
    local dragging2,dragStart2,startPos2,dragInput2=false,nil,nil,nil
    src.InputBegan:Connect(function(inp)
        if State.uiLocked then return end
        if inp.UserInputType==Enum.UserInputType.MouseButton1 or inp.UserInputType==Enum.UserInputType.Touch then
            dragging2=true; dragStart2=inp.Position; startPos2=frame.Position
            inp.Changed:Connect(function()
                if inp.UserInputState==Enum.UserInputState.End then
                    dragging2=false; if onEnd then onEnd(frame.Position) end
                end
            end)
        end
    end)
    src.InputChanged:Connect(function(inp)
        if inp.UserInputType==Enum.UserInputType.MouseMovement or inp.UserInputType==Enum.UserInputType.Touch then dragInput2=inp end
    end)
    UIS.InputChanged:Connect(function(inp)
        if inp==dragInput2 and dragging2 and not State.uiLocked then
            frame.Position=UDim2.new(startPos2.X.Scale,startPos2.X.Offset+(inp.Position.X-dragStart2.X),startPos2.Y.Scale,startPos2.Y.Offset+(inp.Position.Y-dragStart2.Y))
        end
    end)
end

                    for _,part in ipairs(p.Character:GetChildren()) do

                end

    pcall(function()

        rp.FilterDescendantsInstances={c}

            root.CFrame=CFrame.new(res.Position+Vector3.new(0,root.Size.Y/2+State.autoTpDownYOffset,0))
            root.AssemblyLinearVelocity=Vector3.zero
        end
    end)

    if State.autoRightEnabled then
        State.autoRightEnabled=false

        arPhase=1; bigBtns.autoRight.setActive(false)

        if hm then hm:Move(Vector3.zero,false) end

    if startAutoSteal then pcall(startAutoSteal) end
    alConn=RunService.Heartbeat:Connect(function()
        if not State.autoLeftEnabled then return end
        local char=LP.Character; if not char then return end
        local r2=char:FindFirstChild("HumanoidRootPart"); local hm=char:FindFirstChildOfClass("Humanoid")
        if not r2 or not hm or hm.Health<=0 then return end
        local spd=State.normalSpeed
        if alPhase==1 then
            if (Vector3.new(POS.L1.X,r2.Position.Y,POS.L1.Z)-r2.Position).Magnitude<1 then alPhase=2; return end

            if (Vector3.new(POS.L2.X,r2.Position.Y,POS.L2.Z)-r2.Position).Magnitude<1 then
                hm:Move(Vector3.zero,false); r2.Velocity=Vector3.zero; stopAutoLeft()

            hm:Move(mv,false); r2.Velocity=Vector3.new(mv.X*spd,r2.Velocity.Y,mv.Z*spd)
        end

local AIMBOT_SPEED = Cfg.aimbotSpeed or 58
local AIMBOT_RANGE = Cfg.aimbotRange or 120
local HIT_DIST = 5
local SWING_CD = 0.08
local EMA_ALPHA_VEL = 0.35
local EMA_ALPHA_ACC = 0.20
local _emaVel, _emaAcc, _prevVel, _prevTick = {}, {}, {}, {}
local _aimbotConn, _aimbotEquip = nil, nil
local _swingCD = false

local BAT_AIMBOT_NAMES = {"Bat", "Slap", "Iron Slap", "Gold Slap", "Diamond Slap", "Emerald Slap", "Ruby Slap", "Dark Matter Slap", "Flame Slap", "Nuclear Slap", "Galaxy Slap", "Glitched Slap"}

local function findAnyTool()
    local c = LP.Character
    if c then 
        for _, v in ipairs(c:GetChildren()) do 
            if v:IsA("Tool") then return v end 
        end 
    end
    local bp = LP:FindFirstChildOfClass("Backpack")
    if bp then 
        for _, v in ipairs(bp:GetChildren()) do 
            if v:IsA("Tool") then return v end 
        end 
    end
    return nil
end

local function trySwing()
    if _swingCD then return end
    _swingCD = true
    pcall(function()
        local c = LP.Character
        local hm = c and c:FindFirstChildOfClass("Humanoid")
        local tool = findAnyTool()
        if tool and hm then
            if tool.Parent ~= c then 
                hm:EquipTool(tool) 
            end
            local remote = tool:FindFirstChildOfClass("RemoteEvent")
            if remote then 
                remote:FireServer() 
            else 
                tool:Activate() 
            end
        end
    end)
    task.delay(SWING_CD, function() _swingCD = false end)
end

stopAimbot=function()
    State.aimbotActive = false
    if _aimbotConn then 
        _aimbotConn:Disconnect()
        _aimbotConn = nil 
    end
    if _aimbotEquip then 
        task.cancel(_aimbotEquip)
        _aimbotEquip = nil 
    end
    pcall(function()
        local c = LP.Character
        if not c then return end
        local root = c:FindFirstChild("HumanoidRootPart")
        local hm = c:FindFirstChildOfClass("Humanoid")
        if root then root.AssemblyLinearVelocity = Vector3.zero end
        if hm then hm.PlatformStand = false end
    end)
    _swingCD = false
    _emaVel = {}
    _emaAcc = {}
    _prevVel = {}
    _prevTick = {}
    bigBtns.aimbot.setActive(false)
end

startAimbot=function()
    stopConflictingMovement()
    State.aimbotActive = true
    bigBtns.aimbot.setActive(true)
    
    _aimbotConn = RunService.Heartbeat:Connect(function()
        if not State.aimbotActive then 
            stopAimbot()
            return 
        end
        
        local c = LP.Character
        if not c then return end
        
        local root = c:FindFirstChild("HumanoidRootPart")
        if not root then return end
        
        local hm = c:FindFirstChildOfClass("Humanoid")
        if not hm then return end
        
        local best, bestDist = nil, math.huge
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= LP and plr.Character then
                local tr = plr.Character:FindFirstChild("HumanoidRootPart")
                local th = plr.Character:FindFirstChildOfClass("Humanoid")
                if tr and th and th.Health > 0 then
                    local dist = (root.Position - tr.Position).Magnitude
                    if dist < bestDist and dist <= AIMBOT_RANGE then 
                        bestDist = dist
                        best = plr 
                    end
                end
            end
        end
        
        if best and best.Character then
            local targetRoot = best.Character:FindFirstChild("HumanoidRootPart")
            if not targetRoot then return end
            
            local now = tick()
            local rawVel = targetRoot.AssemblyLinearVelocity
            local prevV = _prevVel[best] or rawVel
            local prevT = _prevTick[best] or now
            local dt = math.max(now - prevT, 1/240)
            local rawAcc = (rawVel - prevV) / dt
            
            _emaVel[best] = (_emaVel[best] or rawVel) * (1 - EMA_ALPHA_VEL) + rawVel * EMA_ALPHA_VEL
            _emaAcc[best] = (_emaAcc[best] or rawAcc) * (1 - EMA_ALPHA_ACC) + rawAcc * EMA_ALPHA_ACC
            _prevVel[best] = rawVel
            _prevTick[best] = now
            
            local toTarget = targetRoot.Position - root.Position
            local predTime = math.min(math.max(toTarget.Magnitude / AIMBOT_SPEED, 0.04) * 2.5, 0.40)
            local emaV = _emaVel[best]
            local emaA = _emaAcc[best]
            
            if emaA.Magnitude > 80 then 
                emaA = emaA.Unit * 80 
            end
            
            local predPos = targetRoot.Position + emaV * predTime + emaA * (0.5 * predTime * predTime)
            predPos = Vector3.new(predPos.X, predPos.Y + 0.8, predPos.Z)
            
            local dir = (predPos - root.Position).Unit
            hm.PlatformStand = true
            root.AssemblyLinearVelocity = dir * AIMBOT_SPEED
            
            if bestDist <= HIT_DIST then 
                trySwing() 
            end
        else
            root.AssemblyLinearVelocity = Vector3.zero
            hm.PlatformStand = false
        end
    end)
    
    _aimbotEquip = task.spawn(function()
        while State.aimbotActive do
            pcall(function()
                local tool = findAnyTool()
                local hm = LP.Character and LP.Character:FindFirstChildOfClass("Humanoid")
                if tool and hm and tool.Parent ~= LP.Character then 
                    hm:EquipTool(tool) 
                end
            end)
            task.wait(0.12)
        end
    end)
end

local function getPing()
    local stats = game:GetService("Stats")
    if stats and stats:FindFirstChild("Network") then
        local network = stats.Network
        if network:FindFirstChild("ServerStatsItem") then
            local pingVal = network.ServerStatsItem:FindFirstChild("Total milliseconds between last acknowledgement")
            if pingVal then return pingVal.Value / 1000 end
        end
    end
    return 0.05
end

local function predictTargetPosition(targetHrp, targetHead, pingOffset)
    if not targetHrp or not targetHead then return nil, nil end
    local currentPos = targetHrp.Position
    local velocity = targetHrp.AssemblyLinearVelocity
    local predictedPos = currentPos + (velocity * pingOffset)
    local headPos = targetHead.Position
    local headPredicted = headPos + (velocity * pingOffset)
    return predictedPos, headPredicted
end

local function findBat() local c=LP.Character; if not c then return nil end; local bp=LP:FindFirstChildOfClass("Backpack"); for _,nm in ipairs(BAT_NAMES) do local t=c:FindFirstChild(nm) or (bp and bp:FindFirstChild(nm)); if t then return t end end; return nil end
local _batCounterConns={}
local _batAimLoops={}

local function _fireBatCounter()
    if not State.batCounterEnabled or State.batCounterDebounce then return end
    local char = LP.Character
    if not char then return end
    State.batCounterDebounce = true
    
    local ping = getPing()
    useMedusa()
    
    task.spawn(function()
        local bat = findBat()
        if bat then
            local hm = char:FindFirstChildOfClass("Humanoid")
            if bat.Parent ~= char and hm then
                pcall(function() hm:EquipTool(bat) end)
                task.wait(0.05)
            end
        end
    end)
    
    task.spawn(function()
        local attacker = LP.Character
        if not attacker then return end
        
        local hrp = attacker:FindFirstChild("HumanoidRootPart")
        if not hrp then return end
        
        local targetChar = nil
        local lastHitPlayers = {}
        
        for _, player in pairs(game:GetService("Players"):GetPlayers()) do
            if player ~= LP then
                local pchar = player.Character
                if pchar then
                    table.insert(lastHitPlayers, pchar)
                end
            end
        end
        
        if #lastHitPlayers > 0 then
            targetChar = lastHitPlayers[1]
            for _, pchar in pairs(lastHitPlayers) do
                local phrp = pchar:FindFirstChild("HumanoidRootPart")
                local tCharHrp = targetChar:FindFirstChild("HumanoidRootPart")
                if phrp and tCharHrp then
                    if (phrp.Position - hrp.Position).Magnitude < (tCharHrp.Position - hrp.Position).Magnitude then
                        targetChar = pchar
                    end
                end
            end
            
            local targetHrp = targetChar:FindFirstChild("HumanoidRootPart")
            local targetHead = targetChar:FindFirstChild("Head")
            
            if targetHrp and targetHead then
                local predictedHrpPos, predictedHeadPos = predictTargetPosition(targetHrp, targetHead, ping)
                
                if predictedHrpPos and predictedHeadPos then
                    local posOnPredictedHead = predictedHrpPos + Vector3.new(0, 5, 0)
                    hrp.CFrame = CFrame.new(posOnPredictedHead, predictedHeadPos)
                    hrp.CFrame = hrp.CFrame * CFrame.Angles(math.rad(180), 0, 0)
                    
                    local loopId = "batAim_" .. tostring(os.time() * 1000)
                    local running = true
                    _batAimLoops[loopId] = running
                    
                    task.spawn(function()
                        local duration = 2
                        local elapsed = 0
                        local updateInterval = 0.05
                        
                        while running and elapsed < duration and State.batCounterEnabled do
                            task.wait(updateInterval)
                            
                            if attacker and attacker.Parent and targetChar and targetChar.Parent then
                                local newHrp = attacker:FindFirstChild("HumanoidRootPart")
                                local newTargetHrp = targetChar:FindFirstChild("HumanoidRootPart")
                                local newTargetHead = targetChar:FindFirstChild("Head")
                                
                                if newHrp and newTargetHrp and newTargetHead then
                                    local currentPing = getPing()
                                    local newPredictedHrpPos, newPredictedHeadPos = predictTargetPosition(
                                        newTargetHrp, 
                                        newTargetHead, 
                                        currentPing
                                    )
                                    
                                    if newPredictedHrpPos and newPredictedHeadPos then
                                        local newPosOnHead = newPredictedHrpPos + Vector3.new(0, 5, 0)
                                        newHrp.CFrame = CFrame.new(newPosOnHead, newPredictedHeadPos)
                                        newHrp.CFrame = newHrp.CFrame * CFrame.Angles(math.rad(180), 0, 0)
                                    end
                                end
                            end
                            
                            elapsed = elapsed + updateInterval
                        end
                        
                        _batAimLoops[loopId] = nil
                    end)
                end
            end
        end
    end)
    
    task.wait(0.5)
    State.batCounterDebounce = false
end

local function startBatCounter()
    if Conns.batCounter then return end
    
    applyBatPose(LP.Character)
    
    local function bind(char)
        for _, c2 in pairs(_batCounterConns) do
            pcall(function() c2:Disconnect() end)
        end
        _batCounterConns = {}
        
        if not char then return end
        
        applyBatPose(char)
        
        local hm = char:FindFirstChildOfClass("Humanoid")
        if hm then
            local lastH = hm.Health
            table.insert(_batCounterConns, hm.HealthChanged:Connect(function(nh)
                if nh < lastH - 1 then
                    _fireBatCounter()
                end
                lastH = nh
            end))
        end
    end
    
    bind(LP.Character)
    Conns.batCounter = LP.CharacterAdded:Connect(function(ch)
        task.wait(0.2)
        bind(ch)
    end)
end

local function stopBatCounter()
    resetBatPose(LP.Character)
    
    if Conns.batCounter then
        Conns.batCounter:Disconnect()
        Conns.batCounter = nil
    end
    
    for _, c2 in pairs(_batCounterConns) do
        pcall(function() c2:Disconnect() end)
    end
    _batCounterConns = {}
    
    for loopId, _ in pairs(_batAimLoops) do
        _batAimLoops[loopId] = nil
    end
end

local _stealConn,_stealProgressConn=nil,nil
local _animalCache, _promptCache, _stealCache = {}, {}, {}
local _isStealing = false
local _stealStartTime = 0

local function _isMyPlot(plotName)
    local plot = workspace.Plots and workspace.Plots:FindFirstChild(plotName)
    if not plot then return false end
    local sign = plot:FindFirstChild("PlotSign"); if not sign then return false end
    local yb = sign:FindFirstChild("YourBase")
    return yb and yb:IsA("BillboardGui") and yb.Enabled == true
end

local function _scanPlot(plot)
    if not plot or not plot:IsA("Model") then return end
    if _isMyPlot(plot.Name) then return end
    local podiums = plot:FindFirstChild("AnimalPodiums"); if not podiums then return end
    for _, pod in ipairs(podiums:GetChildren()) do
        if pod:IsA("Model") and pod:FindFirstChild("Base") then
            local uid = plot.Name .. "_" .. pod.Name
            for _, ex in ipairs(_animalCache) do if ex.uid == uid then return end end
            table.insert(_animalCache, {
                name = pod.Name, plot = plot.Name, slot = pod.Name,
                worldPosition = pod:GetPivot().Position, uid = uid,
            })
        end
    end
end

local function _findPromptCached(ad)
    if not ad then return nil end
    local cp = _promptCache[ad.uid]
    if cp and cp.Parent then return cp end
    local plots = workspace:FindFirstChild("Plots"); if not plots then return nil end
    local plot  = plots:FindFirstChild(ad.plot);    if not plot  then return nil end
    local pods  = plot:FindFirstChild("AnimalPodiums"); if not pods then return nil end
    local pod   = pods:FindFirstChild(ad.slot);     if not pod   then return nil end
    local base  = pod:FindFirstChild("Base");        if not base  then return nil end
    local sp    = base:FindFirstChild("Spawn");      if not sp    then return nil end
    local function searchPrompt(obj)
        for _, ch in ipairs(obj:GetDescendants()) do
            if ch:IsA("ProximityPrompt") then return ch end
        end
    end
    local att = sp:FindFirstChild("PromptAttachment")
    local prompt = nil
    if att then
        for _, p in ipairs(att:GetChildren()) do
            if p:IsA("ProximityPrompt") then prompt = p; break end
        end
    end
    if not prompt then prompt = searchPrompt(sp) end
    if prompt then _promptCache[ad.uid] = prompt end
    return prompt
end

local function _buildCallbacks(prompt)
    if _stealCache[prompt] then return end
    local data = { holdCallbacks = {}, triggerCallbacks = {}, ready = true }
    local ok1 = pcall(function() 
        if getconnections then 
            for _, conn in ipairs(getconnections(prompt.PromptButtonHoldBegan)) do
                if type(conn.Function) == "function" then
                    table.insert(data.holdCallbacks, conn.Function)
                end
            end
        end
    end)
    local ok2 = pcall(function()
        if getconnections then
            for _, conn in ipairs(getconnections(prompt.Triggered)) do
                if type(conn.Function) == "function" then
                    table.insert(data.triggerCallbacks, conn.Function)
                end
            end
        end
    end)
    if #data.holdCallbacks > 0 or #data.triggerCallbacks > 0 then
        _stealCache[prompt] = data
    end
end

local function _nearestAnimal()
    local char = LP.Character; if not char then return nil end
    local hrp = char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("UpperTorso")
    if not hrp then return nil end
    local best, bestD = nil, math.huge
    for _, ad in ipairs(_animalCache) do
        if not _isMyPlot(ad.plot) and ad.worldPosition then
            local d = (hrp.Position - ad.worldPosition).Magnitude
            if d < bestD then bestD = d; best = ad end
        end
    end
    return best, bestD
end

local function _execSteal(prompt, animalName)
    local data = _stealCache[prompt]
    if not data or not data.ready then return false end
    data.ready = false; _isStealing = true
    _stealStartTime = tick()
    if _stealProgressConn then _stealProgressConn:Disconnect() end
    
    -- Use new steal bar
    _stealProgressConn = RunService.Heartbeat:Connect(function()
        if not _isStealing then 
            if _stealProgressConn then _stealProgressConn:Disconnect() end
            return 
        end
        local prog = math.clamp((tick() - _stealStartTime) / Steal.StealDuration, 0, 1)
        -- Update new steal bar
        pcall(function()
            if _G.SetStealProgress then
                _G.SetStealProgress(prog * 100)
            end
        end)
    end)
    
    task.spawn(function()
        for _, fn in ipairs(data.holdCallbacks) do task.spawn(fn) end
        local elapsed = 0
        while elapsed < Steal.StealDuration do elapsed = elapsed + task.wait() end
        for _, fn in ipairs(data.triggerCallbacks) do task.spawn(fn) end
        task.wait(0.01)
        if _stealProgressConn then _stealProgressConn:Disconnect(); _stealProgressConn = nil end
        -- Reset steal bar
        pcall(function()
            if _G.SetStealProgress then
                _G.SetStealProgress(0)
            end
        end)
        task.wait(0.01)
        data.ready = true; _isStealing = false
    end)
    return true
end

startAutoSteal=function()
    if _stealConn then return end; State.autoStealEnabled=true; Steal.AutoStealEnabled=true
    local plots = workspace:FindFirstChild("Plots")
    if plots then
        for _, plot in ipairs(plots:GetChildren()) do
            if plot:IsA("Model") then _scanPlot(plot) end
        end
        if not Conns.scanPlots then
            Conns.scanPlots = plots.ChildAdded:Connect(function(plot)
                if plot:IsA("Model") then _scanPlot(plot) end
            end)
        end
    end
    _stealConn = RunService.Heartbeat:Connect(function()
        if not Steal.AutoStealEnabled or _isStealing then return end
        local target, dist = _nearestAnimal()
        if not target then return end
        if dist > Steal.StealRadius then return end
        local prompt = _promptCache[target.uid]
        if not prompt or not prompt.Parent then prompt = _findPromptCached(target) end
        if prompt then _buildCallbacks(prompt); _execSteal(prompt, target.name) end
    end)
end
stopAutoSteal=function()
    State.autoStealEnabled=false; Steal.AutoStealEnabled=false
    if _stealConn then _stealConn:Disconnect(); _stealConn=nil end
    if _stealProgressConn then _stealProgressConn:Disconnect(); _stealProgressConn=nil end
    if Conns.scanPlots then Conns.scanPlots:Disconnect(); Conns.scanPlots=nil end
    _isStealing=false; _stealStartTime=0
    _animalCache={}; _promptCache={}; _stealCache={}
    pcall(function()
        if _G.SetStealProgress then
            _G.SetStealProgress(0)
        end
    end)
end

if LP.Character then pcall(startAutoSteal) else LP.CharacterAdded:Connect(function() task.wait(0.5); pcall(startAutoSteal) end) end

RunService.Heartbeat:Connect(function()
    local c=LP.Character; if not c then return end
    local hm=c:FindFirstChildOfClass("Humanoid"); local hrp=c:FindFirstChild("HumanoidRootPart")
    if not hm or not hrp then return end
    
    local spd = State.normalSpeed
    if State.laggerEnabled then spd = State.laggerSpeed
    elseif State.speedToggled then spd = State.carrySpeed end
    
    pcall(function()
        local head=c:FindFirstChild("Head")
        if head then
            local bb=head:FindFirstChild("SpeedBillboard")
            if bb then
                local lbl=bb:FindFirstChildOfClass("TextLabel")
                if lbl then 
                    local actualSpeed = math.floor(hrp.Velocity.Magnitude + 0.5)
                    local mode = "normal"
                    if State.laggerEnabled then mode = "lagger"
                    elseif State.speedToggled then mode = "carry" end
                    lbl.Text = mode:upper().." | "..tostring(actualSpeed)
                end
            end
        end
    end)
    
    if State.autoLeftEnabled or State.autoRightEnabled or State.aimbotActive then return end
    
    local md = hm.MoveDirection
    if md.Magnitude > 0 then
        hm.WalkSpeed = spd
        local currentVel = hrp.AssemblyLinearVelocity
        local targetVel = Vector3.new(md.X * spd, currentVel.Y, md.Z * spd)
        hrp.AssemblyLinearVelocity = hrp.AssemblyLinearVelocity:Lerp(targetVel, 0.85)
    else
        hm.WalkSpeed = 16
        local currentVel = hrp.AssemblyLinearVelocity
        if currentVel.Magnitude > 1 then
            hrp.AssemblyLinearVelocity = currentVel * 0.9
        else
            hrp.AssemblyLinearVelocity = Vector3.zero
        end
    end
end)
